# Optional Python packages. aranum's core is stdlib-only; these enable specific
# optional capabilities. Pre-stage them on a jump host before going offline:
#   pip install -r requirements-optional.txt
#
# defusedxml — hardens aranumtoolkit/network/nmap-parse.py against XXE /
#              billion-laughs (a hardened stdlib fallback is used without it).
defusedxml
# pywinrm    — required ONLY for aranumtoolkit/network/bulk-enum-windows.py
#              (WinRM transport to Windows hosts).
pywinrm
# impacket   — enables bulk-enum-windows.py's smb transport (wmiexec). Optional;
#              the winrm/ssh transports don't need it.
impacket
# paramiko   — preferred backend for aranumtoolkit/network/ssh-key-triage.py key
#              inventory/fingerprinting (falls back to ssh-keygen if absent).
paramiko>=2.7
# cryptography — paramiko's crypto backend (pulled in by paramiko; listed for
#                offline pre-staging completeness).
cryptography
